Donut Grid (Small Multiples) = VAR _Total = «MEASURE: overall total, used as the denominator for every category's share» VAR _CatA = CALCULATE(«MEASURE», «FILTER: category A») VAR _CatB = CALCULATE(«MEASURE», «FILTER: category B») VAR _CatC = CALCULATE(«MEASURE», «FILTER: category C») VAR _CatD = CALCULATE(«MEASURE», «FILTER: category D») -- Add/remove Cat VARs to match the number of categories (grid is 2 columns — -- an odd count will leave one card alone on the last row, which is fine). VAR _CatAPct = ROUND(DIVIDE(_CatA, _Total, 0) * 100, 1) VAR _CatBPct = ROUND(DIVIDE(_CatB, _Total, 0) * 100, 1) VAR _CatCPct = ROUND(DIVIDE(_CatC, _Total, 0) * 100, 1) VAR _CatDPct = ROUND(DIVIDE(_CatD, _Total, 0) * 100, 1) VAR _Circ = 251.33 -- circumference of the r=40 circle; keep in sync if radius changes -- Each donut here shows ONE category's share vs. the rest (not vs. other -- categories) — so each uses stroke-dashoffset computed from its own % only, -- unlike donut-single where offsets accumulate across segments. VAR _CatAOff = ROUND(_Circ * (1 - _CatAPct/100), 2) VAR _CatBOff = ROUND(_Circ * (1 - _CatBPct/100), 2) VAR _CatCOff = ROUND(_Circ * (1 - _CatCPct/100), 2) VAR _CatDOff = ROUND(_Circ * (1 - _CatDPct/100), 2) RETURN "
