Donut Single (Multi-Segment) = VAR _Total = «MEASURE: overall total» VAR _SegA = CALCULATE(«MEASURE», «FILTER: segment A») VAR _SegB = CALCULATE(«MEASURE», «FILTER: segment B») VAR _SegC = CALCULATE(«MEASURE», «FILTER: segment C») -- Add/remove Seg VARs to match the number of segments (3+ recommended; for -- exactly 2 segments a simpler progress-ring shape may suit better — ask). VAR _SegAPct = ROUND(DIVIDE(_SegA, _Total, 0) * 100, 1) VAR _SegBPct = ROUND(DIVIDE(_SegB, _Total, 0) * 100, 1) VAR _SegCPct = ROUND(DIVIDE(_SegC, _Total, 0) * 100, 1) VAR _Circ = 251.33 -- circumference of the r=40 circle used below; keep in sync if radius changes VAR _SegAArc = ROUND(_Circ * _SegAPct / 100, 2) VAR _SegBArc = ROUND(_Circ * _SegBPct / 100, 2) VAR _SegCArc = ROUND(_Circ * _SegCPct / 100, 2) RETURN "
